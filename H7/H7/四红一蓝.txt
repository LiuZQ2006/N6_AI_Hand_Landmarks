#define PACKET_HEADER  0xAA  

#define PACKET_FOOTER  0xFF  



#define CMD_MOVE_LEFT  0x01

#define CMD_MOVE_RIGHT 0x02

#define CMD_JUMP_UP    0x04

#define CMD_DOWN       0x08

#define CMD_FIST_SKILL 0x10  

#define CMD_ULT_SKILL  0x20  



UART_HandleTypeDef huart1;



/* ================================================================= */

/* 🚀 队友要的“数组”和中断控制标志位 */

/* ================================================================= */

volatile uint8_t rx_byte;          // 中断每次接收1个字节的临时信箱

volatile uint8_t rx_state = 0;     // 状态机进度

volatile uint8_t current_cmd = 0;  // 暂存指令

volatile uint8_t action_flag = 0;  // 动作触发核弹钥匙 (1代表收到完整指令)

volatile uint8_t action_cmd = 0;   // 要执行的具体指令


/* ============ SD卡相关全局变量 ============ */
volatile uint8_t sd_card_ready = 0;  // 1=SD卡就绪，0=不可用
FATFS          sd_fs;                // FATFS文件系统对象
char           sd_path[4];           // 逻辑驱动器路径
/* ========================================= */


void Direct_UART1_Init(void) {

    huart1.Instance = USART1;

    huart1.Init.BaudRate = 115200;

    huart1.Init.WordLength = UART_WORDLENGTH_8B;

    huart1.Init.StopBits = UART_STOPBITS_1;

    huart1.Init.Parity = UART_PARITY_NONE;

    huart1.Init.Mode = UART_MODE_TX_RX;

    huart1.Init.HwFlowCtl = UART_HWCONTROL_NONE;

    huart1.Init.OverSampling = UART_OVERSAMPLING_16;

    if (HAL_UART_Init(&huart1) != HAL_OK) { while(1); }

}



void HAL_UART_MspInit(UART_HandleTypeDef* huart) {

    GPIO_InitTypeDef GPIO_InitStruct = {0};

    if(huart->Instance == USART1) {

        __HAL_RCC_USART1_CLK_ENABLE();

        __HAL_RCC_GPIOA_CLK_ENABLE();

        GPIO_InitStruct.Pin = GPIO_PIN_9 | GPIO_PIN_10;

        GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;

        GPIO_InitStruct.Pull = GPIO_NOPULL;

        GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;

        GPIO_InitStruct.Alternate = GPIO_AF7_USART1; 

        HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);



        /* 【极其关键】：激活 USART1 的硬件中断管家！ */

        HAL_NVIC_SetPriority(USART1_IRQn, 0, 0);

        HAL_NVIC_EnableIRQ(USART1_IRQn);

    }

}

/* ⚡ 中断服务函数：K210发消息时，硬件自动跳到这里执行 */

/* ================================================================= */

void USART1_IRQHandler(void) {

    HAL_UART_IRQHandler(&huart1);

}



/* ⚡ 中断回调函数：专门处理接收到的字节数组 */

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {

    if (huart->Instance == USART1) {

        

        // 纯后台静默组装 3 字节协议包

        if (rx_byte == PACKET_HEADER) {

            rx_state = 1; 

        } 

        else if (rx_state == 1) {

            current_cmd = rx_byte;

            rx_state = 2;

        } 

        else if (rx_state == 2) {

            if (rx_byte == PACKET_FOOTER) {

                // 协议完美对齐！把指令交接给前台主循环，并插上执行钥匙

                action_cmd = current_cmd;

                action_flag = 1;

            }

            rx_state = 0; 

        }

        

        // 【必须保留】：再次开启中断，准备接收下一个字节

        HAL_UART_Receive_IT(&huart1, &rx_byte, 1);

    }

}



/* ⚡ 错误回调：串口出错后自动恢复接收，防止通信永久中断 */

void HAL_UART_ErrorCallback(UART_HandleTypeDef *huart) {

    if (huart->Instance == USART1) {

        HAL_UART_Receive_IT(&huart1, &rx_byte, 1);

    }

}



void LCD_DrawSprite(uint16_t x, uint16_t y, uint16_t width, uint16_t height, const unsigned char *pData) {

    uint16_t color;

    uint32_t index = 0;

    for (uint16_t j = 0; j < height; j++) {

        for (uint16_t i = 0; i < width; i++) {

            color = (pData[index+1] << 8) | pData[index];

            index += 2;

            if (color != 0x07E0) { LCD_DrawPixel(x + i, y + j, color); } 

            else { LCD_DrawPixel(x + i, y + j, LCD_COLOR_BLACK); }

        }

    }

}



void LCD_EraseSprite(uint16_t x, uint16_t y, uint16_t width, uint16_t height) {

    for (uint16_t j = 0; j < height; j++) {

        for (uint16_t i = 0; i < width; i++) { LCD_DrawPixel(x + i, y + j, LCD_COLOR_BLACK); }

    }

}



/* ================================================================ */
/* SD卡逐步诊断函数 v2                                               */
/* 绕过 FATFS_LinkDriver，手动配置驱动表                             */
/* 闪烁N次红灯停顿 = 第N步失败                                       */
/* 蓝灯常亮 = 全部通过                                               */
/* ================================================================ */
void SD_Card_Init_Check(void)
{
    HAL_StatusTypeDef sd_state;
    uint8_t  test_buf[512];
    uint32_t i;

    extern Disk_drvTypeDef disk;

    sd_card_ready = 0;

    /* ---- 第1步：SD卡硬件初始化 ---- */
    sd_state = BSP_SD_Init();
    if (sd_state != HAL_OK)
    {
        while (1) { LED_RED; HAL_Delay(200); LED_RGBOFF; HAL_Delay(500); }
    }
    uSdHandle.Init.ClockDiv = 6;
    HAL_SD_Init(&uSdHandle);
    HAL_SD_ConfigWideBusOperation(&uSdHandle, SDMMC_BUS_WIDE_4B);

    /* ---- 第2步：polling直读扇区0 ---- */
    sd_state = HAL_SD_ReadBlocks(&uSdHandle, test_buf, 0, 1, 5000);
    if (sd_state != HAL_OK)
    {
        while (1) {
            LED_RED; HAL_Delay(200); LED_RGBOFF; HAL_Delay(200);
            LED_RED; HAL_Delay(200); LED_RGBOFF; HAL_Delay(500);
        }
    }

    /* ---- 第3步：手动注册驱动 ---- */
    disk.drv[0] = (Diskio_drvTypeDef*)&SD_Driver;
    disk.lun[0] = 0;
    disk.is_initialized[0] = 1;
    if (disk.nbr < 1) disk.nbr = 1;
    sd_path[0] = '0'; sd_path[1] = ':'; sd_path[2] = '/'; sd_path[3] = 0;

    /* ---- 第3.5步：测试 disk_read 读 test_buf ---- */
    {
        DRESULT dres;
        uint32_t part_start;

        dres = disk_read(0, test_buf, 0, 1);
        if (dres != RES_OK)
        {
            while (1) {
                for (i = 0; i < 3; i++)
                { LED_RED; HAL_Delay(150); LED_RGBOFF; HAL_Delay(150); }
                LED_RED; HAL_Delay(800);
                LED_RGBOFF; HAL_Delay(500);
            }
        }
        /* 从MBR解析分区起始扇区 */
        part_start = *(uint32_t*)(&test_buf[0x1BE + 8]);
        if (part_start == 0) part_start = 0;

        dres = disk_read(0, test_buf, part_start, 1);
        if (dres != RES_OK)
        {
            while (1) {
                for (i = 0; i < 3; i++)
                { LED_RED; HAL_Delay(150); LED_RGBOFF; HAL_Delay(150); }
                HAL_Delay(400);
                LED_BLUE; HAL_Delay(800);
                LED_RGBOFF; HAL_Delay(500);
            }
        }
    }

    /* ---- 第4步：f_mount ---- */
    {
        FRESULT mount_res = f_mount(&sd_fs, "0:", 1);
        if (mount_res != FR_OK)
        {
            while (1)
            {
                for (i = 0; i < 4; i++)
                { LED_RED; HAL_Delay(200); LED_RGBOFF; HAL_Delay(200); }
                HAL_Delay(800);
                for (i = 0; i < (uint32_t)mount_res; i++)
                { LED_BLUE; HAL_Delay(200); LED_RGBOFF; HAL_Delay(200); }
                HAL_Delay(1000);
            }
        }
    }

    /* ---- 第5步：检查 background.bin ---- */
    {
        FIL test_file;
        if (f_open(&test_file, "0:/background.bin", FA_READ) != FR_OK)
        {
            while (1) {
                for (i = 0; i < 5; i++)
                { LED_RED; HAL_Delay(200); LED_RGBOFF; HAL_Delay(200); }
                HAL_Delay(500);
            }
        }
        f_close(&test_file);
    }

    sd_card_ready = 1;
    LED_BLUE;
}

int main(void)

{   

    HAL_Init(); 

    SystemClock_Config();

    LED_GPIO_Config();

  

    Board_MPU_Config(0, MPU_Normal_WT, 0xD0000000, MPU_32MB);

    Board_MPU_Config(1, MPU_Normal_WT, 0x24000000, MPU_512KB);

    SCB_EnableICache();

    SCB_EnableDCache();
    SCB->CACR |= 1 << 2;   /* 强制DCache为write-through，和野火官方SD例程一致 */



    Direct_UART1_Init();       

    GTP_Init_Panel(); 

    LCD_Init();

    LCD_LayerInit(0, LCD_FB_START_ADDRESS, RGB888);

    LCD_LayerInit(1, LCD_FB_START_ADDRESS+(LCD_GetXSize()*LCD_GetYSize()*4), RGB888);

    LCD_DisplayOn(); 



    LCD_SelectLayer(0);

    LCD_Clear(LCD_COLOR_BLACK);  

    LCD_SelectLayer(1);

    LCD_Clear(LCD_COLOR_TRANSPARENT);



    LCD_SetTransparency(0, 0);

    LCD_SetTransparency(1, 255);

    Delay(0xfff);


    /* 🔍 检测SD卡是否就绪（结果存入 sd_card_ready） */
    SD_Card_Init_Check();


    int x_pos = 295;

    int y_pos = 90;

    int last_x = 295;

    int last_y = 90;

    int move_speed = 5; 

    

    int last_width = 213; 

    uint8_t frame_step = 0;

    const unsigned char* current_frame = gImage_walframe1code;



    LCD_DrawSprite(x_pos, y_pos, 213, 299, current_frame);

    

    /* 🚀 启动第一次硬件中断接收监听！ */

    HAL_UART_Receive_IT(&huart1, &rx_byte, 1);

    

    while(1)

    {       

        // 如果底层中断捕获到了完整的 3 字节协议包，钥匙就会变成 1

        if (action_flag == 1) {

            

            uint8_t is_moving = 0; 

            

            switch(action_cmd) {

                case CMD_MOVE_LEFT:  x_pos -= move_speed; is_moving = 1; break; 

                case CMD_MOVE_RIGHT: x_pos += move_speed; is_moving = 1; break; 

                case CMD_JUMP_UP:    y_pos -= move_speed; is_moving = 1; break; 

                case CMD_DOWN:       y_pos += move_speed; is_moving = 1; break; 

                

                case CMD_FIST_SKILL:

                {

                    if (x_pos > (800 - 311)) x_pos = 800 - 311;

                    const unsigned char* atk_frames[] = {

                        gImage_attackframe1, gImage_attackframe2, gImage_attackframe3,

                        gImage_attackframe4, gImage_attackframe5, gImage_attackframe4,

                        gImage_attackframe3, gImage_attackframe2, gImage_attackframe1

                    };

                    for(int i = 0; i < 9; i++) {

                        LCD_EraseSprite(x_pos, y_pos, last_width, 299); 

                        LCD_DrawSprite(x_pos, y_pos, 311, 299, atk_frames[i]); 

                        last_width = 311; 

                        Delay(0x6FFFFF); 

                    }

                    LCD_EraseSprite(x_pos, y_pos, last_width, 299);

                    LCD_DrawSprite(x_pos, y_pos, 213, 299, gImage_walframe1code);

                    last_width = 213;

                    break;

                }

                

                case CMD_ULT_SKILL:

                {

                    const unsigned char* ult_frames[] = {

                        gImage_Ultframe1, gImage_Ultframe2, gImage_Ultframe3, gImage_Ultframe4

                    };

                    for(int loop = 0; loop < 4; loop++) { 

                        for(int i = 0; i < 4; i++) {

                            LCD_EraseSprite(x_pos, y_pos, last_width, 299);

                            LCD_DrawSprite(x_pos, y_pos, 213, 299, ult_frames[i]);

                            last_width = 213;

                            Delay(0x6FFFFF); 

                        }

                    }

                    LCD_EraseSprite(x_pos, y_pos, last_width, 299);

                    LCD_DrawSprite(x_pos, y_pos, 213, 299, gImage_walframe1code);

                    last_width = 213;

                    break;

                }

                default: break; 

            }

            

            if (x_pos < 0) x_pos = 0;

            if (y_pos < 0) y_pos = 0;

            if (x_pos > (800 - 213)) x_pos = 800 - 213;

            if (y_pos > (480 - 299)) y_pos = 480 - 299;



            if (is_moving && (x_pos != last_x || y_pos != last_y)) {

                frame_step++;

                if (frame_step > 5) frame_step = 0;

                

                switch(frame_step) {

                    case 0: current_frame = gImage_walframe1code; break;

                    case 1: current_frame = gImage_walframe2code; break;

                    case 2: current_frame = gImage_walframe3code; break;

                    case 3: current_frame = gImage_walframe4code; break;

                    case 4: current_frame = gImage_walframe3code; break;

                    case 5: current_frame = gImage_walframe2code; break;

                }

                LCD_EraseSprite(last_x, last_y, last_width, 299);

                LCD_DrawSprite(x_pos, y_pos, 213, 299, current_frame);

                

                last_x = x_pos;

                last_y = y_pos;

                last_width = 213; 

            }

            

            // 动作执行完毕，拔出钥匙，等待下一次中断触发

            action_flag = 0; 

        }

    }

}



void SystemClock_Config(void) {

  RCC_OscInitTypeDef RCC_OscInitStruct = {0};

  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  HAL_PWREx_ConfigSupply(PWR_LDO_SUPPLY);

  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE0);

  while(!__HAL_PWR_GET_FLAG(PWR_FLAG_VOSRDY)) {}

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSE;

  RCC_OscInitStruct.HSEState = RCC_HSE_ON;

  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;

  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;

  RCC_OscInitStruct.PLL.PLLM = 5;

  RCC_OscInitStruct.PLL.PLLN = 192;

  RCC_OscInitStruct.PLL.PLLP = 2;

  RCC_OscInitStruct.PLL.PLLQ = 4;

  RCC_OscInitStruct.PLL.PLLR = 2;

  RCC_OscInitStruct.PLL.PLLRGE = RCC_PLL1VCIRANGE_2;

  RCC_OscInitStruct.PLL.PLLVCOSEL = RCC_PLL1VCOWIDE;

  RCC_OscInitStruct.PLL.PLLFRACN = 0;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK) { while(1); }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK|RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2|RCC_CLOCKTYPE_D3PCLK1|RCC_CLOCKTYPE_D1PCLK1;

  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;

  RCC_ClkInitStruct.SYSCLKDivider = RCC_SYSCLK_DIV1;

  RCC_ClkInitStruct.AHBCLKDivider = RCC_HCLK_DIV2;

  RCC_ClkInitStruct.APB3CLKDivider = RCC_APB3_DIV2;

  RCC_ClkInitStruct.APB1CLKDivider = RCC_APB1_DIV2;

  RCC_ClkInitStruct.APB2CLKDivider = RCC_APB2_DIV2;

  RCC_ClkInitStruct.APB4CLKDivider = RCC_APB4_DIV2;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK) { while(1); }

}



void Delay(__IO uint32_t nCount) { for(; nCount != 0; nCount--); }
